	id = tfv_raj.50
	title = tfv_raj.50.t
	desc = tfv_raj.50.desc
	picture = GFX_report_event_bose_meeting
	
	is_triggered_only = yes

	fire_only_once = yes

	immediate = {
		RAJ = {
			OVERLORD = {
				save_event_target_as = Britain
			}
		}
	}

	#Help
	option = {
		name = tfv_raj.50.a
		ai_chance = {
			base = 50
			modifier = {
				factor = 0.2
				JAP_can_help_against_ENG = no				
			}
			modifier = {
				factor = 4
				OR = {
					has_opinion = {
						target = RAJ
						value > 75
					}
					has_opinion = {
						target = ENG
						value < 0
					}
				}

			}
			modifier = {
				factor = 5
				ROOT_will_help_RAJ = yes
			}			
			modifier = {
				factor = 4	
				JAP = {
					fascism > 0.5
				}
			}
		}	
		set_global_flag = RAJ_seek_help_from_japan_promised
		if = {
			limit = {
				RAJ = { NOT = { has_war_together_with = ENG } }
				NOT = { has_war_with = ENG }
			}
			ENG = { country_event = { id = tfv_raj.51 hours = 6 } }
			else = {
				#Answer event - Accept
				RAJ = { country_event = { id = tfv_raj.52 } }
			}
		}
	}

	#No
	option = {
		name = tfv_raj.50.b
		ai_chance = {
			base = 5
			modifier = {
				factor = 4
				ROOT_will_help_RAJ = no
			}
		}
		#Answer event - Reject
		RAJ = { country_event = { id = tfv_raj.53 } }
	}
}
