country_event = {
	id = POR_rio_saga.1
	title = POR_rio_saga.1.t
	desc = POR_rio_saga.1.desc
	picture = GFX_report_event_sailors_in_working_rig

	is_triggered_only = yes

	option = { #RIO IS MINE
		name = POR_rio_saga.1.a

		ai_chance = {

		}

		custom_effect_tooltip =  POR_rio_saga.1.a_tt
		random_country = {
			limit = { controls_state = 500 }
			country_event = { id = POR_rio_saga.2 days = 70 }
		}

	option = { #lets be friends instead
		name = POR_rio_saga.1.b

		ai_chance = {
			base = 30
			modifier = {
				has_completed_focus = POR_the_fifth_empire
				factor = 0
			}
		}
		add_stability = 0.1
		add_political_power = 50
		random_country = {
			limit = { controls_state = 500 }
			add_opinion_modifier = {
				target = ROOT
				modifier = BUL_strengthened_relations_with_us
			}
			ROOT = {
				add_opinion_modifier = {
					target = PREV
					modifier = BUL_strengthened_relations_with_us
				}
			}
			if = {
				limit = {
					ROOT = { has_government = fascism }
				}
				add_timed_idea = { idea = fascist_influence days = 1095 }
			}
			add_ai_strategy = {
				type = alliance
				id = ROOT
				value = 200
			}
		}
	}
}
