	OR = {

			has_country_flag = AUS_lands_of_old_nap_offered_flag
			has_country_flag = AUS_lands_of_old_bribery_done_flag

		}

			has_country_flag = AUS_lands_of_old_border_exercises_done_flag
			has_country_flag = AUS_lands_of_old_border_guards_done_flag
		}
		AND = { #NAP refused, and threats are off-limits on diplomatic paths
			has_country_flag = AUS_lands_of_old_nap_offered_flag

		}
	}
