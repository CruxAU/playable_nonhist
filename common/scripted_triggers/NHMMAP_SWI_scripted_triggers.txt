SWI_AI_is_pro_allies = { #AI tries to stay in strong cantons, then slowly moves to balanced, hires democrats and some socialists
	OR = {		
		has_game_rule = {
			rule = SWI_AI_BEHAVIOR
			option = NEUTRALITY_PRO_ALLIES
		}
		has_country_flag = SWI_AI_RANDOM_NEUTRALITY_PRO_ALLIES
	}
}
